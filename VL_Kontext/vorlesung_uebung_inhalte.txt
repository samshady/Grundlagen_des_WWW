module Main exposing (main)

import Browser
import Html exposing (Attribute, Html, a, button, div, fieldset, footer, h1, h2, header, img, input, label, li, nav, node, p, section, span, table, tbody, td, text, th, thead, tr, ul)
import Html.Attributes exposing (checked, class, coords, href, id, name, placeholder, shape, src, style, title, type_, usemap, value)
import Html.Events exposing (onClick, onInput)
import List
import String


main : Program () { modal : Maybe ModalState, saved : List Shape } Msg
main =
    Browser.sandbox
        { init = { modal = Nothing, saved = [] }
        , view = view
        , update = update
        }


type alias Model =
    { modal : Maybe ModalState
    , saved : List Shape
    }


type Shape
    = Circle String String String



--  | Rectangle String String String String
--  | Polygon (List ( ( String, String ) ))


type ModalState
    = ShowTable
    | InputShape Shape
    | ShowShape Int Shape


type ModalMsg
    = OpenShowTable
    | OpenInputShape
    | Change Shape
    | Add Shape
    | OpenShapeMessage Int Shape


type Msg
    = OpenModal ModalMsg
    | CloseModal


update : Msg -> Model -> Model
update msg model =
    case msg of
        OpenModal modalMsg ->
            updateModal modalMsg model

        CloseModal ->
            { model | modal = Nothing }


updateModal : ModalMsg -> Model -> Model
updateModal modalMsg model =
    case modalMsg of
        OpenShowTable ->
            { model | modal = Just ShowTable }

        OpenInputShape ->
            { model | modal = Just (InputShape (Circle "" "" "")) }

        Change shape ->
            { model | modal = Just (InputShape shape) }

        Add shape ->
            { model | modal = Nothing, saved = shape :: model.saved }

        OpenShapeMessage i s ->
            { model | modal = Just (ShowShape i s) }


view : Model -> Html Msg
view model =
    div []
        [ applicationHeader
        , buttons
        , clickableImage model.saved
        , viewModal model
        ]


applicationTitle : String.String
applicationTitle =
    "Formen über der Informatik"


applicationSubTitle : String.String
applicationSubTitle =
    "Eingeben, Anzeigen, Finden und Clicken"


applicationHeader : Html Msg
applicationHeader =
    div []
        [ h1 [] [ text applicationTitle ]
        , h2 [] [ text applicationSubTitle ]
        ]


buttons : Html Msg
buttons =
    div []
        [ a [ onClick (OpenModal OpenInputShape) ] [ text "Neue Form" ]
        , a [ onClick (OpenModal OpenShowTable) ] [ text "Tabelle mit Formen" ]
        ]


clickableImage : List Shape -> Html Msg
clickableImage shapeList =
    div []
        [ img
            [ src "http://www.informatik.uni-halle.de/im/1285058520_1381_00_800.jpg"
            , usemap "#informatik"
            ]
            []
        , imgMap [ name "informatik" ]
            (shapes2areas shapeList)
        ]


shapes2areas : List Shape -> List (Html Msg)
shapes2areas shapes =
    List.indexedMap
        (\i s ->
            let
                ( gebietstyp, gebietskoordinaten ) =
                    case s of
                        Circle x y r ->
                            ( "circ", x ++ "," ++ y ++ "," ++ r )
            in
            area
                [ shape gebietstyp
                , coords gebietskoordinaten
                , href ("#s" ++ String.fromInt i)
                , onClick (OpenModal (OpenShapeMessage i s))
                , title ("Shape " ++ String.fromInt i)
                ]
                []
        )
        shapes


shapes2rows : List Shape -> List (Html Msg)
shapes2rows shapes =
    List.indexedMap
        (\i s ->
            let
                ( gebietstyp, gebietskoordinaten ) =
                    case s of
                        Circle x y r ->
                            ( "Kreis", x ++ "," ++ y ++ "," ++ r )
            in
            tr [ id ("s" ++ String.fromInt i) ]
                [ td [] [ text (String.fromInt i) ]
                , td [] [ text gebietstyp ]
                , td [] [ text gebietskoordinaten ]
                ]
        )
        shapes


viewModal : Model -> Html Msg
viewModal model =
    case model.modal of
        Nothing ->
            span [] []

        Just modalState ->
            div []
                [ case modalState of
                    ShowTable ->
                        showShapesAsTableInModal model.saved

                    InputShape shape ->
                        inputShapeInModal shape

                    ShowShape i s ->
                        showShape i s
                ]


modalHeader : String -> Html Msg
modalHeader title =
    a [ onClick CloseModal ] []


modalFooter : List (Html Msg) -> Html Msg
modalFooter modalButtons =
    div []
        (modalButtons
            ++ [ a [ onClick CloseModal ] [ text "Schliessen" ] ]
        )


showShapesAsTableInModal : List Shape -> Html Msg
showShapesAsTableInModal shapes =
    div []
        [ modalHeader "Tabelle der Formen"
        , table []
            [ thead []
                [ tr []
                    [ th [] [ text "Laufende Nummer" ]
                    , th [] [ text "Gebietstyp" ]
                    , th [] [ text "Gebietskoordinaten" ]
                    ]
                ]
            , tbody [] (shapes2rows shapes)
            ]
        , modalFooter []
        ]


inputShapeInModal : Shape -> Html Msg
inputShapeInModal s =
    div []
        [ inputShape s
        , modalFooter
            [ a [ onClick (OpenModal (Add s)) ] [ text "Form hinzufügen" ] ]
        ]


inputShape : Shape -> Html Msg
inputShape s =
    case s of
        Circle x y r ->
            let
                changeCircleX newX =
                    OpenModal (Change (Circle newX y r))

                changeCircleY newY =
                    OpenModal (Change (Circle x newY r))

                changeCircleR newR =
                    OpenModal (Change (Circle x y newR))
            in
            fieldset []
                [ div []
                    [ text "X-Position"
                    , input [ class "input", type_ "text", value x, onInput changeCircleX ] []
                    ]
                , div []
                    [ text "Y-Position"
                    , input [ class "input", type_ "text", value y, onInput changeCircleY ] []
                    ]
                , div []
                    [ text "Radius"
                    , input [ class "input", type_ "text", value r, onInput changeCircleR ] []
                    ]
                ]


showShape : Int -> Shape -> Html Msg
showShape i s =
    div []
        [ modalHeader ("Shape " ++ String.fromInt i)
        , text ("Shape " ++ String.fromInt i ++ " wurde gefunden!")
        , modalFooter []
        ]


imgMap : List (Attribute msg) -> List (Html msg) -> Html msg
imgMap attributes children =
    node "map" attributes children


area : List (Attribute msg) -> List (Html msg) -> Html msg
area attributes children =
    node "area" attributes children


ariaLabel : String -> Attribute msg
ariaLabel value =
    Html.Attributes.attribute "aria-label" value

Aufgabe 4.1 Button für das interaktive Zeichnen von Formen mit Dropdown-Menue
Bevor es an das interaktive Zeichnen geht, soll ein Button mit Dropdown-Menue erstellt werden. Verwenden Sie den entsprechenden Dropdown-Button von Bulma. In dem Dropdown-Menu sollen Kreis und Rechteck als Formen auswählbar sein. Wenn eine Form ausgewählt ist, dann sollen die nächsten Mouse-Clicks anderes interpretiert werden: (a) für Kreis soll mit dem ersten Klick der Mittelpunkt und mit dem zweiten Klick der Radius festgelegt werden, (b) für Rechtecke soll der erste Klick eine Ecke und der zweite Klick die Ausdehnung mit Breite und Höhe festlegen. Nach der Auswahl soll das Dropdown-Menue dann wieder ausgeblendet werden und der Button selbst soll "Kreis zeichnen: abbrechen" bzw. "Rechteck zeichnen: abbrechen" heißen und diese Aktion bewirken.

Um zwischen den verschiedenen Zwischenzuständen zu wechseln, kann der Msg-Type erweitert werden

type Msg
   = OpenModal ModalMsg
   | CloseModal
   | AddShapeButton AddShapeButtonMsg

type AddShapeButtonMsg
    = HideAddShapeDropdownMenu
    | ShowAddShapeDropdownMenu
    | MouseDraw Ishape
    | MouseClick Ishape

type Ishape
    = Icircle String String (Maybe String)
    | Irectangle String String (Maybe ( String, String ))
[/code]

Die verschieden Zustände können durch Erweiterung des Models um ein Feld addShapeButton vom Type AddShapeButtonState umgesetzt werden:
[code]
type alias Model =
    { modal : Maybe ModalState
    , saved : List Shape
    , addShapeButton : AddShapeButtonState
    }

type AddShapeButtonState
    = ShowButtonMenu
    | HideButtonMenu
    | DrawShape Ishape
Erweitern Sie die update-Funktion analog zu updateModal um eine updateAddShapeButton-Funktion, die sich um die Auswertung der AddShapeButtonMsg kümmert.

Mit dem Dropdown-Button von Bulma müssen mehre Mouse-Click-Ereignisse verbunden werden: (a) äußerer Button zum Anzeigen und Verstecken des Menüs und (b) die inneren Menü-Punkte. Die entsprechenden DOM-Elemente sind in einer Vorfahren-Nachfahren-Beziehung. Lesen Sie über den Fluss von Ereignissen im DOM in Selfhtml nach. Ereignisse wie Clicks werden ebenfalls bei den Vorfahren des eigentlichen Elements ausgelöst. Dies ist hier hinderlich, weil Klicks im Menü ebenfalls einen Klick auf den Button auslösen und so beide Aktionen ausgeführt werden. Um das Aufsteigen des Ereignisses im DOM-Baum zu verhindern, muss in Elm die neue Eventhandler-Funktion onClickNoBubblingUp definiert werden und anstelle von onClick verwendet werden.

onClickNoBubblingUp : msg -> Attribute msg
onClickNoBubblingUp msg =
    Html.Events.stopPropagationOn "click" (Json.Decode.map (\m -> ( m, True )) (Json.Decode.succeed msg))
In diesem Code-Schnipsel wird noch die Bibliothek elm/json verwendet, die Sie in ellie unter packages einbinden müssen oder wenn Sie direkt mit elm entwickeln durch das Installieren des Packages elm/json

elm install elm/json
Die Bibliothek muss dann am Anfang des Elm-Codes eingebunden werden:

import Json.Decode
Als Ergebnis dieser Aufgabe haben Sie einen funktionieren Dropdown-Menu-Button. Die Auswahl von Kreis oder Rechteck im Menü braucht noch nicht das interaktive Zeichnen auslösen, sollte aber das Menü wieder schließen.

Aufgabe 4.2 Das responsive Darstellen des Bildes und der Formen als SVG-Grafik
Bisher wurden die Regionen mittels veralteter HTML-Elemente dargestellt. In dieser Aufgabe sollen sie innerhalb eines SVG-Elements das Informatik-Bild einbinden und die bisher eingegeben Formen als SVG-Elemente durchsichtig anzeigen und anklickbar machen. Ein Klick auf eine Form soll wie bisher auch ein Modal mit einer Nachricht öffnen.

Bulma stellt mit [figure einen Container für responsive Bilder]https://bulma.io/documentation/elements/image/ bereit, in den sie ein SVG-Element einbinden können. Damit sich das SVG-Bild responsive an Breite des Bildschirms anpasst, darf nur die Breite mit 100% angegeben werden. Das Bild ist etwa 800x600. Die viewbox definiert das interne Koordinatensystem innerhalb des SVG.Bildes.

<section class="section">
  <div class="container">
     <figure class="image">
        <svg width="100%" viewBox="0 0 800 600" version="1.1">
          <image width="800" height="600" title="Informatikgebäude" xlink:href="http://www.informatik.uni-halle.de/im/1285058520_1381_00_800.jpg"></image>
        </svg>
     </figure>
  </div>
</section>
Die SVG-Elemente für Kreise und Rechtecke sollen mit der Klasse shape gestylt werden. Die Pseudo-Klasse hover wird automatisch vom Browser entsprechend der Mausposition vergeben:

.shape { fill: black;
         stroke: red;
         stroke-width: 3;
         fill-opacity:0;
      }
.shape:hover {
        fill: white;
        opacity:0.5;
        fill-opacity:0.5;
}
Achten Sie bei der Klassenzuweisen von SVG-Elementen darauf, dass Sie die Funktion SVG.Attributes.class nutzen und nicht die Funktion HTML.Attributes.class. Das passiert, wenn Sie HTML.Attributes.class mit expose in den Scope aufnehmen und führt zu einer Exception.

Die Elemente für Kreise und Rechtecke (siehe https://wiki.selfhtml.org/wiki/SVG/Elemente für Kreise und Rechteck in SVG) müssen im DOM nach dem Bild als Kinder innerhalb des SVG-Elements eingefügt werden. Dafür können Sie Funktion für das Erzeugen der Area-Elemente umarbeiten.

Als Ergebnis sollen alle bisher eingegebenen Formen durchsichtig mit rotem Rand über dem Bild des Informatikgebäudes dargestellt werden und anklickbar sein.

Aufgabe 4.3. Das interaktive Eingeben von Kreisen und Rechtecken
Um Kreise und Rechtecke interaktiv eingeben zu können, müssen Maus-Positionen in interpretierbaren Koordinaten abgefragt. Das MouseEvent-Interface des Browser stellt die Eigenschaften offsetX und offsetY bzw. clientX und clientY bereit, die den Pixel-Abstand des Mauszeiger in der jeweiligen Koordinate zur Padding-Kante der Zielknotens im DOM bereitstellen. Diese Koordinaten müssen noch in die Koordinaten des SVG-Koordinatensystems umgerechnet werden. Die in diesem Ellie definierten Hilfsfunktionen stellen eine Lösung bereit, die in allen Browser funktioniert, siehe https://ellie-app.com/yHvCdR3CTkGa1 .

Um den Pixel-Abstand zu sinnvollen Koordinaten zu machen, muss der Zielknoten genauso groß wie die Viewbox SVG-Elements selbst sein, also 800x600 Pixel. Weiterhin muss der Zielknoten der oberste sichtbare Element innerhalb des SVG-Elements sein. Ein Ansatz wäre, ein durchsichtiges Rechteck als letztes Kind des SVG-Elements einzufügen, wenn interaktiv eine Form eingegeben werden soll.

<rect width="800" height="600" x="0" y="0" style="fill:blue;stroke:pink;stroke-width:5;fill-opacity:0.1;stroke-opacity:0.9"></rect>
Unter diesem durchsichtigen Rechteck, d.h. an Positionen vor dem Rechteck in der Kind-Liste des SVG-Elements, können die bisher eingegebenen Formen und die aktuell einzugebende Form dargestellt werden.

Als Ergebnis dieser Aufgabe können Kreise und Rechtecke interaktiv eingegeben werden.
